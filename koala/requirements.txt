# Python packages koala needs on top of the BSP. Install into a --system-site-packages venv so the
# BSP's own numpy, onnxruntime, tflite_runtime, cv2 and gi stay visible:
#
#     python3 -m venv --system-site-packages ~/koala/venv       (install.sh does this)
#
# The vision half of the pilot needs NOTHING from pip -- `./run.sh --no-voice` works on a bare board.
# Everything below is for the voice and VLM half, and is deliberately much shorter than NXP's
# pyproject.toml, which pulls the RAG and LLM stacks we do not ship.

# Wake word needs nothing at all beyond the BSP's numpy.

# shared_utils/utils.so imports colorama at module scope, so STT and TTS both need it even though
# nothing we write is coloured.
colorama==0.4.6

# Text-to-speech: weight decryption, resampling inside tts/utils.so, plus WAV reading for the
# earcons. scipy is not in NXP's own dependency list -- it arrives there via scikit-learn, which is
# part of the RAG stack we do not ship, so we have to name it ourselves.
pycryptodome==3.23.0
scipy
soundfile==0.13.1

# Speech-to-text: silero-vad drags in torch and torchaudio; transformers supplies the tokenizer.
silero-vad==5.1.2
torch==2.8.0
torchaudio==2.8.0
transformers==5.3.0

# Vision-language model ("describe scene"). SmolVLM weights come from Hugging Face on first use --
# `./run.sh --prefetch` pulls them ahead of time.
torchvision==0.23.0
accelerate==1.7.0

# /IOTCONNECT: telemetry, C2D commands and the S3 snapshot upload. Pure Python -- paho-mqtt for the
# connection and (via the aws-s3 extra) boto3 for the upload; no native build, nothing to conflict
# with the BSP's numpy/onnxruntime/tflite_runtime. The extra is what makes `snapshot` upload rather
# than only save, and the SDK degrades to a clear error message without it.
iotconnect-sdk-lite[aws-s3]==1.3.0

# KVS WebRTC: the live view of the display. aiortc is pure Python; av, cryptography and pylibsrtp
# arrive as aarch64 wheels, so there is no compiler involved. Note what is NOT happening here --
# the i.MX95's hardware H.264 encoder does the compressing inside GStreamer, and aiortc only
# packetises what it is handed (see webrtc.py). av is present to carry those bytes in an av.Packet,
# not to encode anything, and libav's own encoders are never called.
aiortc==1.15.0
av
websockets

# The LLM on the Ara-240 ('ask'). Two pure-Python packages and nothing else: the model, torch and
# optimum-ara all live in connector/venv, in a separate process, behind an HTTP port -- which is the
# only reason this is installable here at all (optimum-ara pins its own torch and would fight the
# BSP stack above). See app/ask.py and connector/README.md.
strands-agents==1.50.2
openai==2.52.0

# Deliberately NOT installed:
#   onnxruntime     the BSP build exposes NeutronExecutionProvider; pip's would break Neutron
#   tflite_runtime  BSP's is ABI-matched to the Neutron delegate; pip's is not
#   cv2             the BSP's bundles YuNet + SFace, which ARE this pilot's face models
#   numpy           BSP has 2.2.3, satisfying eIQ's >=2.2.3,<3.0.0
#   ai-edge-litert  BSP has 1.2.0, satisfying eIQ's >=1.2.0
#   pyalsaaudio     we drive the GStreamer backend; this is only the ALSA fallback, and it is a
#                   source distribution needing ALSA headers the image does not have
#   posix_ipc PyQt6 only for NXP's GUI, which we do not ship
#   sentence-transformers scikit-learn sentencepiece   RAG and LLM only, both out of scope
#
# Requires OpenCV >= 4.5.4 in the BSP for the face classes. Verify on-device first:
#     python3 diag-face.py face_detection_yunet_2023mar.onnx face_recognition_sface_2021dec.onnx
